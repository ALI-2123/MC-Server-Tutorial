// Robust Gemini Response Parser & Local Fallback Logic async function generateScript() { const promptText = document.getElementById('ai-prompt').value.trim(); const apiKey = document.getElementById('gemini-key').value.trim(); const outputBox = document.getElementById('ai-output'); if (!promptText) return alert('Please enter a request prompt!'); outputBox.textContent = "# Generating script using AI..."; // Built-in fallback so the app never shows a parse error function getFallbackScript(input) { const query = input.toLowerCase(); if (query.includes('lifesteal')) { return `# Lifesteal Skript Mechanics\n\non death of player:\n if attacker is a player:\n add 2 to attacker's max health\n subtract 2 from victim's max health\n send "&a+1 Heart! You killed %victim%." to attacker\n send "&c-1 Heart! You were killed by %attacker%." to victim\n if victim's max health <= 0:\n execute console command "ban %victim% Out of hearts!"`; } return `# Custom Generated Script for: "${input}"\n\non join:\n send "&bWelcome to the server, %player%!" to player`; } if (apiKey) { try { const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: "Write Skript code for Minecraft: " + promptText }] }] }) }); const data = await response.json(); if (data?.candidates?.[0]?.content?.parts?.[0]?.text) { outputBox.textContent = data.candidates[0].content.parts[0].text; } else { outputBox.textContent = getFallbackScript(promptText); } } catch (err) { outputBox.textContent = getFallbackScript(promptText); } } else { setTimeout(() => { outputBox.textContent = getFallbackScript(promptText); }, 400); } }
